set -euo pipefail

ARCHIVE="$HOME/Downloads/MMRI_FINAL_20_OF_20_FULL_INTEGRATION_2026-10-26.tar.gz"
OLD="$HOME/MMRI_TEST_20_20/MMRI_BACKEND_20_STEP_WORK"
BACKUP="$HOME/MMRI_TEST_20_20/MMRI_BACKUP_BEFORE_FINAL_20_20_$(date +%Y%m%d_%H%M%S)"

if [ ! -f "$ARCHIVE" ]; then
  ARCHIVE="$(find "$HOME/Downloads" "$HOME" -maxdepth 2 -type f -name 'MMRI_FINAL_20_OF_20_FULL_INTEGRATION_2026-10-26.tar.gz' -print -quit 2>/dev/null || true)"
fi
[ -n "${ARCHIVE:-}" ] && [ -f "$ARCHIVE" ] || { echo "FINAL ARCHIVE NOT FOUND. Put it in $HOME/Downloads"; exit 1; }

export NVM_DIR="$HOME/.nvm"
if [ -s "$NVM_DIR/nvm.sh" ]; then
  . "$NVM_DIR/nvm.sh"
  nvm install 22.16.0
  nvm use 22.16.0
else
  node -e 'const v=process.versions.node;if(!v.startsWith("22.16.")){console.error("Node 22.16.x is required. Install/use Node 22.16.0 first.");process.exit(1)}'
fi

mkdir -p "$HOME/MMRI_TEST_20_20"
if [ -d "$OLD" ]; then
  mv "$OLD" "$BACKUP"
  echo "Previous source saved at: $BACKUP"
fi

mkdir -p "$OLD"
tar -xzf "$ARCHIVE" -C "$OLD" --strip-components=1
cd "$OLD"

sha256sum -c FINAL_20_OF_20_SHA256_MANIFEST.txt
chmod +x VERIFY_FINAL_20_OF_20_SOURCE.sh INSTALL_FINAL_20_OF_20_DEPENDENCIES_AND_BUILD.sh RUN_FINAL_20_OF_20_LOCAL.sh START_LOCAL.sh STOP_LOCAL.sh 2>/dev/null || true

./INSTALL_FINAL_20_OF_20_DEPENDENCIES_AND_BUILD.sh

printf '\nFINAL 20/20 INSTALLED. Starting local acceptance test...\n'
./RUN_FINAL_20_OF_20_LOCAL.sh
